cmake_minimum_required(VERSION 3.24)
project(RevisionSDK LANGUAGES C CXX ASM_MASM RC)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Revision SDK requires the Windows MSVC x86 toolchain. Use ./build.ps1 to initialize it.")
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
if(CMAKE_CONFIGURATION_TYPES OR NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release|Publish)$")
    message(FATAL_ERROR "Use a single-configuration generator (Ninja) with Debug, Release, or Publish.")
endif()

# CMake initializes the custom configuration's flags to empty.
if(CMAKE_BUILD_TYPE STREQUAL "Publish")
    foreach(kind C CXX ASM_MASM RC EXE_LINKER MODULE_LINKER SHARED_LINKER STATIC_LINKER)
        if(CMAKE_${kind}_FLAGS_PUBLISH STREQUAL "")
            set(CMAKE_${kind}_FLAGS_PUBLISH "${CMAKE_${kind}_FLAGS_RELEASE}")
        endif()
    endforeach()
endif()

set(REVISION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(REVISION_SRC "${REVISION_ROOT}/sp/src")
set(REVISION_GAMES EZ2 CACHE STRING "Game to build (EZ2 only)")
set_property(CACHE REVISION_GAMES PROPERTY STRINGS EZ2)
set(REVISION_BUILD_GROUP game CACHE STRING "Default targets: game, shaders, maptools, tools, everything")
set_property(CACHE REVISION_BUILD_GROUP PROPERTY STRINGS game shaders maptools tools everything)
set(REVISION_STEAMWORKS_VERSION 1261 CACHE STRING "Steamworks header version as an integer")
option(REVISION_BUILD_EXPANDED_STEAM "Rebuild Expanded Steam (requires matching updated Steamworks headers)" OFF)
option(REVISION_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(REVISION_USE_PCH "Use game DLL precompiled headers" ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Match the SDK's legacy C++ ABI and exception model rather than CMake's /EHsc defaults.
string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
include(cmake/SourceSDK.cmake)
include(cmake/DiscordRPC.cmake)

if(NOT REVISION_BUILD_GROUP MATCHES "^(game|shaders|maptools|tools|everything)$")
    message(FATAL_ERROR "Unknown REVISION_BUILD_GROUP: ${REVISION_BUILD_GROUP}")
endif()
if(NOT REVISION_GAMES STREQUAL "EZ2")
    message(FATAL_ERROR "REVISION_GAMES must be EZ2 (received '${REVISION_GAMES}').")
endif()

foreach(lib tier1 mathlib raytrace vgui_controls vscript responserules fgdlib)
    revision_add_target(${lib} STATIC)
endforeach()
if(REVISION_BUILD_EXPANDED_STEAM)
    revision_add_target(expanded_steam STATIC)
endif()

set(ez2_bin "${REVISION_ROOT}/sp/game/mod_ez2/bin")
revision_add_runtime_target(client_ez2 SHARED "${ez2_bin}" OUTPUT_NAME client
    RUNTIME_DEPENDENCIES discord-rpc)
revision_add_runtime_target(server_ez2 SHARED "${ez2_bin}" OUTPUT_NAME server)
if(REVISION_USE_PCH)
    target_precompile_headers(client_ez2_binary PRIVATE "${REVISION_SRC}/game/client/cbase.h")
    target_precompile_headers(server_ez2_binary PRIVATE "${REVISION_SRC}/game/server/cbase.h")
endif()
add_custom_target(game DEPENDS client_ez2 server_ez2)

set(runtime_directory "${REVISION_ROOT}/sp/game/bin")
add_custom_target(maptools)
foreach(tool vbsp vrad_dll vrad_launcher vvis_dll vvis_launcher)
    string(REGEX REPLACE "_launcher$" "" output_name "${tool}")
    if(tool MATCHES "_dll$")
        revision_add_runtime_target(${tool} SHARED "${runtime_directory}" OUTPUT_NAME "${output_name}")
    else()
        revision_add_runtime_target(${tool} EXECUTABLE "${runtime_directory}" OUTPUT_NAME "${output_name}")
    endif()
    add_dependencies(maptools ${tool})
endforeach()

add_custom_target(tools)
foreach(tool captioncompiler glview height2normal motionmapper tgadiff vice vtf2tga vtfdiff)
    revision_add_runtime_target(${tool} EXECUTABLE "${runtime_directory}")
    add_dependencies(tools ${tool})
endforeach()
set_target_properties(glview_binary PROPERTIES WIN32_EXECUTABLE TRUE)
revision_add_runtime_target(serverplugin_empty SHARED "${runtime_directory}")
add_dependencies(tools serverplugin_empty maptools)

# LUX shader generation is opt-in.
if(REVISION_BUILD_GROUP MATCHES "^(shaders|everything)$")
    include(cmake/LuxShaders.cmake)
    revision_add_lux_shaders()
endif()
if(REVISION_BUILD_GROUP STREQUAL "everything")
    add_custom_target(everything DEPENDS game tools shaders)
endif()
add_custom_target(revision ALL DEPENDS ${REVISION_BUILD_GROUP})
message(STATUS "Revision SDK: game=EZ2, group=${REVISION_BUILD_GROUP}, configuration=${CMAKE_BUILD_TYPE}")
